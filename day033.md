### 思路：
- 相当于一个有选择的全排列，在不存在斜线相连的时候加入ans，判断逻辑有两个写法：1.专门写个判断函数，对于输入r和c，
  遍历所有已经存过的的行列，如果存在与r+c或r-c相等，则判断在斜线上。
  ```python
  class Solution(object):
    def solveNQueens(self, n):
        """
        :type n: int
        :rtype: List[List[str]]
        """
        ans = []
        col = [-2] * n

        def valid(r, c):
            for R in range(r):
                C = col[R]
                if R + C == r + c or R - C == r - c:
                    return False
            return True

        def dfs(r, s):
            if r == n:
                ans.append(['.'*c + 'Q' + '.'*(n - 1 - c) for c in col])
                return
            
            for c in s:
                if valid(r, c):
                    col[r] = c
                    dfs(r + 1, s - {c})
        
        dfs(0, set(range(n)))
        return ans

- 使用布尔数组记录r+c和r-c是否已经存在，若存在，则说明在斜线上。
  ```python
  class Solution(object):
    def solveNQueens(self, n):
        """
        :type n: int
        :rtype: List[List[str]]
        """
        ans = []
        col = [-2] * n
        m = 2*n-1
        dict1 = [False]*m
        dict2 = [False]*m


        def dfs(r, s):
            if r == n:
                ans.append(['.'*c + 'Q' + '.'*(n - 1 - c) for c in col])
                return
            
            for c in s:
                if dict1[r + c] == False and dict2[r - c] == False:
                    col[r] = c
                    dict1[r + c] = True
                    dict2[r - c] = True
                    dfs(r + 1, s - {c})
                    dict1[r + c] = False
                    dict2[r - c] = False                  
        
        dfs(0, set(range(n)))
        return ans
